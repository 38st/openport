cmake_minimum_required(VERSION 3.25)
project(openport_packages LANGUAGES NONE)

foreach(required OPENPORT_STAGE OPENPORT_PACKAGE_VERSION OPENPORT_PACKAGE_ARCH)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Set ${required}")
  endif()
endforeach()
if(NOT OPENPORT_PACKAGE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Expected a release version")
endif()
if(OPENPORT_PACKAGE_ARCH STREQUAL "amd64")
  set(CPACK_RPM_PACKAGE_ARCHITECTURE x86_64)
elseif(OPENPORT_PACKAGE_ARCH STREQUAL "arm64")
  set(CPACK_RPM_PACKAGE_ARCHITECTURE aarch64)
else()
  message(FATAL_ERROR "Expected amd64 or arm64")
endif()
foreach(file bin/openportd bin/openport-probe share/openport/web/index.html)
  if(NOT EXISTS "${OPENPORT_STAGE}/${file}")
    message(FATAL_ERROR "Missing staged release file: ${file}")
  endif()
endforeach()

install(PROGRAMS "${OPENPORT_STAGE}/bin/openportd" "${OPENPORT_STAGE}/bin/openport-probe" DESTINATION bin)
install(DIRECTORY "${OPENPORT_STAGE}/share/openport/web" DESTINATION share/openport)
install(DIRECTORY "${OPENPORT_STAGE}/docs/" DESTINATION share/doc/openport)
install(FILES "${OPENPORT_STAGE}/README.md" "${OPENPORT_STAGE}/LICENSE" DESTINATION share/doc/openport)
install(FILES openport.service DESTINATION lib/systemd/system)

set(CPACK_GENERATOR "DEB;RPM")
set(CPACK_PACKAGE_NAME openport)
set(CPACK_PACKAGE_VERSION "${OPENPORT_PACKAGE_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "openport-${OPENPORT_PACKAGE_VERSION}-linux-${OPENPORT_PACKAGE_ARCH}")
set(CPACK_PACKAGE_VENDOR "OpenPort contributors")
set(CPACK_PACKAGE_CONTACT "https://github.com/38st/openport/issues")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/38st/openport")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Options analytics and paper-trading simulator")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "${OPENPORT_PACKAGE_ARCH}")
# Built in Ubuntu 24.04; do not imply compatibility with older libc versions.
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.39), libstdc++6 (>= 13), libssl3t64, zlib1g, libzstd1, ca-certificates")
set(CPACK_DEBIAN_PACKAGE_SECTION finance)
set(CPACK_RPM_PACKAGE_LICENSE MIT)
set(CPACK_RPM_PACKAGE_RELOCATABLE OFF)
set(CPACK_RPM_PACKAGE_REQUIRES "glibc >= 2.39, libstdc++ >= 13, openssl-libs >= 3, zlib, libzstd, ca-certificates")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/lib/systemd /usr/lib/systemd/system)
include(CPack)
